cmake_minimum_required(VERSION 3.24)
project(ConflictBench VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Concurrent Test)
add_library(cbcore src/core.cpp src/core.h)
target_link_libraries(cbcore PUBLIC Qt6::Core)
target_include_directories(cbcore PUBLIC src)
if(WIN32)
  target_link_libraries(cbcore PRIVATE advapi32)
endif()
add_library(cbui src/workbench.cpp src/workbench.h)
target_link_libraries(cbui PUBLIC cbcore Qt6::Widgets Qt6::Concurrent)
target_compile_definitions(cbui PUBLIC CB_VERSION="${PROJECT_VERSION}")
qt_add_executable(ConflictBench MACOSX_BUNDLE WIN32 src/main.cpp)
target_link_libraries(ConflictBench PRIVATE cbui)
set_target_properties(ConflictBench PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER "io.github.rad1092.conflictbench" MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}" MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}" MACOSX_BUNDLE_BUNDLE_NAME "ConflictBench")
foreach(target cbcore cbui ConflictBench)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE cbcore Qt6::Test)
  add_test(NAME core_safety COMMAND core_tests)
  add_test(NAME desktop_smoke COMMAND ConflictBench --smoke-test)
  set_tests_properties(desktop_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/gui_tests.cpp")
    add_executable(gui_tests tests/gui_tests.cpp)
    target_link_libraries(gui_tests PRIVATE cbui Qt6::Test)
    add_test(NAME desktop_workflow COMMAND gui_tests)
    set_tests_properties(desktop_workflow PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  endif()
endif()
if(UNIX AND NOT APPLE)
  set_target_properties(ConflictBench PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
endif()
include(GNUInstallDirs)
install(TARGETS ConflictBench BUNDLE DESTINATION . RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES LICENSE THIRD_PARTY_NOTICES.md README.md DESTINATION share/conflictbench)
install(DIRECTORY docs/ DESTINATION share/conflictbench/docs)
